# Start : Not Gated curriculum 0 -> -5
# Note: exact initial 0.0 is not feasible with this multiplicative curriculum because the scale would remain stuck at 0.

    exp:g1-23dof-fast-sac-quiet-not-gated \
    simulator:isaacgym \
    logger:wandb \
    --logger.entity robox \
    --logger.project holosoma \
    --logger.video.enabled False \
    --logger.name 0_iteration_not_gated_quiet_curriculum_0_to_5_ \
    --training.seed 1 \
    --training.num-envs 4096 \
    --reward.terms.penalty-feet-landing-speed-not-gated.tags '["quiet_penalty_curriculum"]' \
    --reward.terms.penalty-feet-landing-speed-not-gated.weight=-5.0 \
    --curriculum.setup-terms.penalty-curriculum.params.tag=quiet_penalty_curriculum \
    --curriculum.setup-terms.penalty-curriculum.params.initial-scale=0.01 \
    --curriculum.setup-terms.penalty-curriculum.params.min-scale=0.0 \
    --curriculum.setup-terms.penalty-curriculum.params.max-scale=1.0 \
